import { z } from "zod"
import { defineArtifactContract } from "@/milo/contract"

const timestamp = z.iso.datetime()
const identifier = z.string().max(1024)
const fingerprint = z.string().regex(/^[0-9a-f]{64}$/)
const meetingKey = z.string().regex(/^mb:[0-9a-f]{32}$/)
const label = z.string().max(160)
const note = z.string().max(300)
const gap = z.string().max(200)
const url = z.string().max(500)

const attendee = z.strictObject({
  name: z.string().max(100).optional(),
  organization: z.string().max(100).optional(),
  role: z.string().max(100).optional(),
  response: z.enum(["accepted", "tentative", "declined", "unknown"]).optional(),
})

const evidencePoint = z.strictObject({
  text: note,
  kind: z.enum(["fact", "inference", "recommendation"]),
  area: z.enum([
    "purpose",
    "agenda",
    "person",
    "history",
    "commitment",
    "development",
    "decision",
    "talking-point",
    "question",
    "risk",
  ]),
  sourceIds: z.array(z.string().max(60)).max(4),
})

const source = z.strictObject({
  id: z.string().max(60),
  kind: z.string().max(40),
  label,
  occurredAt: timestamp.optional(),
  retrievedAt: timestamp,
  url: url.optional(),
})

const researchAttempt = z.strictObject({
  attemptId: z.string().max(60),
  eventFingerprint: fingerprint,
  status: z.enum(["pending", "ready", "partial", "sparse", "failed"]),
  startedAt: timestamp,
  completedAt: timestamp.optional(),
  findings: z.array(evidencePoint).max(12),
  sources: z.array(source).max(6),
  gaps: z.array(gap).max(4),
})

const briefing = z.strictObject({
  purpose: evidencePoint.optional(),
  outcome: note.optional(),
  essentials: z.array(evidencePoint).max(3),
  details: z.array(evidencePoint).max(8),
  unknowns: z.array(gap).max(4),
})

const receipt = z.strictObject({
  revision: z.number().int().nonnegative(),
  eventStartsAt: timestamp,
  deliveredAt: timestamp,
  destination: z.enum(["email", "slack-dm"]),
})

const event = z.strictObject({
  provider: z.string().max(60),
  calendarId: identifier.optional(),
  eventId: identifier,
  seriesId: identifier.optional(),
  title: z.string().max(200),
  startsAt: timestamp,
  endsAt: timestamp,
  status: z.string().max(40),
  fingerprint,
  organizer: attendee.optional(),
  attendees: z.array(attendee).max(8),
  location: z.string().max(200).optional(),
})

const meeting = z.strictObject({
  event,
  selectedBecause: z.string().max(200),
  status: z.enum([
    "queued",
    "researching",
    "ready",
    "partial",
    "sparse",
    "failed",
    "stale",
    "cancelled",
  ]),
  revision: z.number().int().nonnegative(),
  preparedAt: timestamp.optional(),
  preparedForFingerprint: fingerprint.optional(),
  briefing: briefing.optional(),
  sources: z.array(source).max(6),
  delivery: z
    .strictObject({ morning: receipt.optional(), reminder: receipt.optional() })
    .optional(),
})

export type MeetingBriefing = z.infer<typeof meeting>
export type BriefingPoint = z.infer<typeof evidencePoint>

export const contract = defineArtifactContract({
  version: 2,
  state: {
    briefings: {
      key: "meeting-briefing/briefings",
      scope: "shared",
      description:
        "Canonical Meeting Briefing state. Patch one meeting branch at a time and preserve unrelated meetings.",
      schema: z.strictObject({
        schemaVersion: z.literal(2),
        timezone: z.string().max(100),
        scan: z.strictObject({
          scannedAt: timestamp,
          horizonStart: timestamp,
          horizonEnd: timestamp,
          status: z.enum(["ready", "partial", "failed"]),
          gaps: z.array(gap).max(4),
        }),
        meetings: z.record(meetingKey, meeting),
      }),
      schemaName: "MeetingBriefings",
      schemaVersion: 2,
    },
    research: {
      key: "meeting-briefing/research",
      scope: "personal",
      description:
        "Private, temporary Meeting Briefing research. Clear each attempt after synthesis.",
      schema: z.strictObject({
        schemaVersion: z.literal(2),
        attempts: z.record(meetingKey, researchAttempt),
      }),
      schemaName: "MeetingBriefingResearch",
      schemaVersion: 2,
    },
  },
})
