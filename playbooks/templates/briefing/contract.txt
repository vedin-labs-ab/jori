import { z } from "zod"
import { defineAppContract } from "@/milo/contract"

const timestamp = z.iso.datetime()
const identifier = z.string().max(512)
const contentHash = z.string().regex(/^[0-9a-f]{32}$/)
const meetingKey = z.string().regex(/^[0-9a-f]{32}$/)
const dispatchKey = z.string().min(1).max(120)
const label = z.string().max(120)
const note = z.string().max(240)
const gap = z.string().max(160)
const url = z.string().max(400)

const attendee = z.strictObject({
  name: z.string().max(80).optional(),
  organization: z.string().max(80).optional(),
  role: z.string().max(80).optional(),
  response: z.enum(["accepted", "tentative", "declined", "unknown"]).optional(),
})

const evidencePoint = z.strictObject({
  text: note,
  kind: z.enum(["fact", "inference", "recommendation"]),
  area: z.enum([
    "purpose",
    "agenda",
    "person",
    "history",
    "commitment",
    "development",
    "decision",
    "talking-point",
    "question",
    "risk",
  ]),
  sourceIds: z.array(z.string().max(60)).min(1).max(3),
})

const source = z.strictObject({
  id: z.string().max(60),
  kind: z.string().max(40),
  label,
  occurredAt: timestamp.optional(),
  retrievedAt: timestamp,
  url: url.optional(),
})

const briefing = z.strictObject({
  purpose: evidencePoint.optional(),
  outcome: note.optional(),
  essentials: z.array(evidencePoint).max(3),
  details: z.array(evidencePoint).max(5),
  unknowns: z.array(gap).max(4),
})

const receipt = z.strictObject({
  revision: z.number().int().nonnegative(),
  eventStartsAt: timestamp,
  deliveredAt: timestamp,
  destination: z.enum(["email", "slack-dm"]),
})

const dispatch = z.strictObject({
  kind: z.enum(["morning", "reminder", "manual", "failure"]),
  status: z.enum(["sending", "delivered", "failed", "unknown"]),
  attemptedAt: timestamp,
  updatedAt: timestamp,
  deliveredAt: timestamp.optional(),
  destination: z.enum(["email", "slack-dm"]),
  error: gap.optional(),
})

const event = z.strictObject({
  provider: z.string().max(60),
  calendarId: identifier.optional(),
  eventId: identifier,
  seriesId: identifier.optional(),
  title: z.string().max(200),
  startsAt: timestamp,
  endsAt: timestamp,
  status: z.string().max(40),
  contentHash,
  organizer: attendee.optional(),
  attendees: z.array(attendee).max(6),
  location: z.string().max(200).optional(),
})

const meeting = z.strictObject({
  event,
  whyItMatters: z.string().max(160),
  status: z.enum([
    "queued",
    "researching",
    "ready",
    "partial",
    "sparse",
    "failed",
    "stale",
    "cancelled",
  ]),
  revision: z.number().int().nonnegative(),
  preparedAt: timestamp.optional(),
  preparedForContentHash: contentHash.optional(),
  briefing: briefing.optional(),
  sources: z.array(source).max(4),
  delivery: z
    .strictObject({ morning: receipt.optional(), reminder: receipt.optional() })
    .optional(),
})

export type MeetingBriefing = z.infer<typeof meeting>
export type BriefingPoint = z.infer<typeof evidencePoint>

export const contract = defineAppContract({
  version: 5,
  state: {
    briefings: {
      key: "meeting-briefing/briefings",
      scope: "shared",
      description:
        "Your prepared briefings, one per meeting: the research behind each one, when it was delivered, and the calendar scan that found it.",
      usage:
        "Canonical briefing state, keyed by each calendar event's entityKey. Patch one meeting branch at a time and preserve unrelated meetings. revision counts user-visible briefing changes and receipts record the delivered revision; meetings age out seven local days after they end, dispatches after seven days.",
      schema: z.strictObject({
        schemaVersion: z.literal(5),
        timezone: z.string().max(100),
        scan: z.strictObject({
          scannedAt: timestamp,
          horizonStart: timestamp,
          horizonEnd: timestamp,
          status: z.enum(["ready", "partial", "failed"]),
          gaps: z.array(gap).max(6),
        }),
        meetings: z.record(meetingKey, meeting),
        dispatches: z.record(dispatchKey, dispatch),
      }),
      schemaName: "MeetingBriefings",
      schemaVersion: 5,
    },
  },
})
