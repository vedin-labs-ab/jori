import { CalendarDays, ChevronLeft } from "lucide-react"
import { useEffect, useMemo, useState } from "react"
import { Badge } from "@/components/ui/badge"
import { Button } from "@/components/ui/button"
import {
  Empty,
  EmptyDescription,
  EmptyHeader,
  EmptyMedia,
  EmptyTitle,
} from "@/components/ui/empty"
import { Skeleton } from "@/components/ui/skeleton"
import { useJoriState } from "@/jori"
import { BriefingView } from "./Briefing"
import { contract, type MeetingBriefing } from "./contract"

type MeetingEntry = [string, MeetingBriefing]

export default function App() {
  const state = useJoriState(contract.state.briefings)
  const [chosenId, setChosenId] = useState(readMeetingId)
  const now = useNow()
  const meetings = useMemo(() => orderedMeetings(state.value?.meetings, now), [state.value, now])
  const nextId = findNextId(meetings, now)
  const selected = selectMeeting(meetings, chosenId, nextId)
  const timezone = safeTimezone(state.value?.timezone)
  const emptyFailure = state.error !== null || state.value?.scan.status === "failed"
  const emptyIncomplete = state.value?.scan.status === "partial"
  const deliveryIssue = latestDeliveryIssue(state.value?.dispatches)

  useEffect(() => {
    const readHash = () => setChosenId(readMeetingId())
    window.addEventListener("hashchange", readHash)
    return () => window.removeEventListener("hashchange", readHash)
  }, [])

  function choose(id: string | null) {
    setChosenId(id)
    writeMeetingId(id)
  }

  return (
    <main className="mx-auto min-h-screen max-w-6xl p-4 sm:p-6 lg:p-10">
      <Header
        checkedAt={state.value?.scan.scannedAt}
        count={upcomingCount(meetings, now)}
        timezone={timezone}
      />
      {deliveryIssue === undefined ? null : (
        <p className="mt-5 rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive" role="alert">
          {deliveryIssue.error ?? "The latest briefing delivery could not be confirmed. Jori will not resend it automatically, to avoid a duplicate."}
        </p>
      )}
      {state.error === null || meetings.length === 0 ? null : (
        <p className="mt-5 rounded-lg border border-destructive/30 bg-destructive/5 p-3 text-sm text-destructive" role="alert">
          {state.error.message}
        </p>
      )}
      {state.value === null || state.value.scan.status === "ready" || meetings.length === 0 ? null : (
        <p className="mt-5 rounded-lg border bg-muted/40 p-3 text-sm text-muted-foreground" role="status">
          The latest calendar check was {state.value.scan.status}. {state.value.scan.gaps[0] ?? "Some meetings may be missing."}
        </p>
      )}
      {state.status === "loading" ? (
        <Loading />
      ) : meetings.length === 0 && (emptyFailure || emptyIncomplete) ? (
        <IncompleteState failed={emptyFailure} gap={state.value?.scan.gaps[0]} />
      ) : meetings.length === 0 ? (
        <EmptyState />
      ) : (
        <div className="mt-6 overflow-hidden rounded-xl border bg-background md:grid md:grid-cols-[18rem_minmax(0,1fr)]">
          <aside className={chosenId === null ? "block" : "hidden md:block"}>
            <MeetingList
              meetings={meetings}
              nextId={nextId}
              onChoose={choose}
              selectedId={selected?.[0]}
              timezone={timezone}
            />
          </aside>
          <section className={chosenId === null ? "hidden min-w-0 md:block" : "min-w-0"}>
            {selected === undefined ? null : (
              <>
                <div className="border-b p-3 md:hidden">
                  <Button onClick={() => choose(null)} size="sm" variant="ghost">
                    <ChevronLeft /> All meetings
                  </Button>
                </div>
                <BriefingView meeting={selected[1]} timezone={timezone} />
              </>
            )}
          </section>
        </div>
      )}
    </main>
  )
}

function Header({ checkedAt, count, timezone }: { checkedAt?: string; count: number; timezone?: string }) {
  return (
    <header className="flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
      <div>
        <p className="text-sm text-muted-foreground">Prepared context for the meetings that matter</p>
        <h1 className="text-2xl font-semibold tracking-tight">Meeting Briefing</h1>
      </div>
      <div className="flex items-center gap-2 text-sm text-muted-foreground">
        <Badge variant="secondary">{count} upcoming</Badge>
        {checkedAt === undefined ? null : <span>Checked {formatRelative(checkedAt, timezone ?? "UTC")}</span>}
      </div>
    </header>
  )
}

function MeetingList({ meetings, nextId, onChoose, selectedId, timezone }: {
  meetings: MeetingEntry[]
  nextId?: string
  onChoose: (id: string) => void
  selectedId?: string
  timezone: string
}) {
  return (
    <nav aria-label="Meeting briefings" className="max-h-[calc(100vh-12rem)] overflow-y-auto">
      {meetings.map(([id, meeting]) => (
        <button
          className={`w-full border-b p-4 text-left transition-colors last:border-b-0 hover:bg-muted/50 ${id === selectedId ? "bg-muted" : ""}`}
          key={id}
          onClick={() => onChoose(id)}
          type="button"
        >
          <div className="flex items-center justify-between gap-2 text-xs text-muted-foreground">
            <span>{formatMeetingTime(meeting.event.startsAt, timezone)}</span>
            <span className="flex items-center gap-2">
              {id === nextId ? <Badge variant="secondary">Next</Badge> : null}
              <Status status={meeting.status} />
            </span>
          </div>
          <p className="mt-1 truncate font-medium">{meeting.event.title}</p>
          <p className="mt-1 truncate text-sm text-muted-foreground">{attendeeLine(meeting)}</p>
        </button>
      ))}
    </nav>
  )
}

function Status({ status }: { status: MeetingBriefing["status"] }) {
  const label = status === "researching" ? "Researching" : status[0].toUpperCase() + status.slice(1)
  return <span>{label}</span>
}

function Loading() {
  return (
    <div className="mt-6 grid gap-4 rounded-xl border p-5 md:grid-cols-[16rem_1fr]">
      <Skeleton className="h-72" />
      <div className="space-y-4"><Skeleton className="h-8 w-2/3" /><Skeleton className="h-24" /><Skeleton className="h-40" /></div>
    </div>
  )
}

function EmptyState() {
  return (
    <Empty className="mt-6 min-h-72 border">
      <EmptyHeader>
        <EmptyMedia variant="icon"><CalendarDays /></EmptyMedia>
        <EmptyTitle>No meeting needs a briefing yet</EmptyTitle>
        <EmptyDescription>Jori will add one here when an upcoming meeting has useful context to prepare.</EmptyDescription>
      </EmptyHeader>
    </Empty>
  )
}

function IncompleteState({ failed, gap }: { failed: boolean; gap?: string }) {
  return (
    <Empty className="mt-6 min-h-72 border">
      <EmptyHeader>
        <EmptyMedia variant="icon"><CalendarDays /></EmptyMedia>
        <EmptyTitle>{failed ? "Meeting check did not finish" : "Meeting check incomplete"}</EmptyTitle>
        <EmptyDescription>{gap ?? "Jori could not verify the upcoming calendar, so no new briefings were created."}</EmptyDescription>
      </EmptyHeader>
    </Empty>
  )
}

function useNow() {
  const [now, setNow] = useState(Date.now)

  useEffect(() => {
    const interval = window.setInterval(() => setNow(Date.now()), 30_000)
    return () => window.clearInterval(interval)
  }, [])

  return now
}

function orderedMeetings(meetings: Record<string, MeetingBriefing> | undefined, now: number): MeetingEntry[] {
  return Object.entries(meetings ?? {}).sort(([, left], [, right]) => {
    const leftGroup = meetingGroup(left, now)
    const rightGroup = meetingGroup(right, now)
    if (leftGroup !== rightGroup) return leftGroup - rightGroup
    return leftGroup === 0
      ? left.event.startsAt.localeCompare(right.event.startsAt)
      : right.event.startsAt.localeCompare(left.event.startsAt)
  })
}

function meetingGroup(meeting: MeetingBriefing, now: number) {
  if (meeting.status === "cancelled") return 2
  return new Date(meeting.event.endsAt).getTime() >= now ? 0 : 1
}

function findNextId(meetings: MeetingEntry[], now: number) {
  return meetings.find(([, meeting]) => meeting.status !== "cancelled" && new Date(meeting.event.endsAt).getTime() >= now)?.[0]
}

function selectMeeting(meetings: MeetingEntry[], chosenId: string | null, nextId?: string) {
  return meetings.find(([id]) => id === chosenId) ?? meetings.find(([id]) => id === nextId) ?? meetings[0]
}

function upcomingCount(meetings: MeetingEntry[], now: number) {
  return meetings.filter(([, meeting]) => meeting.status !== "cancelled" && new Date(meeting.event.endsAt).getTime() >= now).length
}

function attendeeLine(meeting: MeetingBriefing) {
  return meeting.event.attendees.map((attendee) => attendee.name ?? attendee.organization).filter(Boolean).join(", ") || "Attendees not listed"
}

function latestDeliveryIssue(dispatches: Record<string, { status: string; updatedAt: string; error?: string }> | undefined) {
  return Object.values(dispatches ?? {})
    .filter((dispatch) => dispatch.status === "failed" || dispatch.status === "unknown")
    .sort((left, right) => right.updatedAt.localeCompare(left.updatedAt))[0]
}

function formatMeetingTime(value: string, timezone: string) {
  return new Intl.DateTimeFormat(undefined, { weekday: "short", month: "short", day: "numeric", hour: "numeric", minute: "2-digit", timeZone: timezone }).format(new Date(value))
}

function formatRelative(value: string, timezone: string) {
  return new Intl.DateTimeFormat(undefined, { month: "short", day: "numeric", hour: "numeric", minute: "2-digit", timeZone: timezone }).format(new Date(value))
}

function safeTimezone(value: string | undefined) {
  if (value === undefined) return "UTC"
  try {
    new Intl.DateTimeFormat(undefined, { timeZone: value }).format()
    return value
  } catch {
    return "UTC"
  }
}

function readMeetingId() {
  return new URLSearchParams(window.location.hash.replace(/^#/, "")).get("m")
}

function writeMeetingId(id: string | null) {
  const params = new URLSearchParams(window.location.hash.replace(/^#/, ""))
  if (id === null) params.delete("m")
  else params.set("m", id)
  window.history.replaceState(null, "", `${window.location.pathname}${window.location.search}#${params.toString()}`)
}
