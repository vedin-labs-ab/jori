import { useMemo, useState } from "react"
import { Badge } from "@/components/ui/badge"
import { Button } from "@/components/ui/button"
import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "@/components/ui/card"
import { Input } from "@/components/ui/input"
import { Separator } from "@/components/ui/separator"
import { useMiloState } from "@/milo"
import { contract } from "./contract"

type Dossier = NonNullable<
  ReturnType<typeof useDossiers>["value"]
>["days"][string][string]

export default function App() {
  const dossiers = useDossiers()
  const initial = useMemo(readLocation, [])
  const [day, setDay] = useState(initial.day)
  const [meetingId, setMeetingId] = useState<string | null>(initial.meetingId)
  const meetings = Object.entries(dossiers.value?.days[day] ?? {}).sort(
    ([, left], [, right]) => left.startsAt.localeCompare(right.startsAt)
  )
  const meeting = meetingId === null ? undefined : dossiers.value?.days[day]?.[meetingId]

  if (meeting !== undefined) {
    return (
      <main className="mx-auto grid max-w-3xl gap-5 p-6 md:p-10">
        <Button className="w-fit" variant="ghost" onClick={() => setMeetingId(null)}>
          Back to {day}
        </Button>
        <DossierView dossier={meeting} />
      </main>
    )
  }

  return (
    <main className="mx-auto grid max-w-4xl gap-6 p-6 md:p-10">
      <header className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
        <div>
          <p className="text-sm text-muted-foreground">Prepared context for the meetings that matter</p>
          <h1 className="text-2xl font-semibold tracking-tight">Meeting prep</h1>
        </div>
        <Input
          aria-label="Meeting date"
          className="w-auto"
          type="date"
          value={day}
          onChange={(event) => setDay(event.target.value)}
        />
      </header>
      {dossiers.status === "loading" ? (
        <p className="text-sm text-muted-foreground">Loading dossiers…</p>
      ) : meetings.length === 0 ? (
        <Card>
          <CardHeader>
            <CardTitle>No meeting prep for this day</CardTitle>
            <CardDescription>Dossiers appear here when a meeting is worth preparing for.</CardDescription>
          </CardHeader>
        </Card>
      ) : (
        <div className="grid gap-3">
          {meetings.map(([id, dossier]) => (
            <button className="text-left" key={id} type="button" onClick={() => setMeetingId(id)}>
              <Card className="transition-colors hover:bg-muted/40">
                <CardHeader>
                  <div className="flex items-start justify-between gap-4">
                    <div>
                      <CardDescription>{formatTime(dossier.startsAt)}</CardDescription>
                      <CardTitle>{dossier.title}</CardTitle>
                    </div>
                    {isNext(dossier.startsAt, meetings) ? <Badge variant="secondary">Next</Badge> : null}
                  </div>
                </CardHeader>
                <CardContent className="text-muted-foreground">
                  {dossier.attendees.map(attendeeName).join(", ") || "No attendees listed"}
                </CardContent>
              </Card>
            </button>
          ))}
        </div>
      )}
      {dossiers.error === null ? null : <p className="text-sm text-destructive">{dossiers.error.message}</p>}
    </main>
  )
}

function DossierView({ dossier }: { dossier: Dossier }) {
  return (
    <Card>
      <CardHeader>
        <CardDescription>{formatDateTime(dossier.startsAt)} · Prepared {formatDateTime(dossier.preparedAt)}</CardDescription>
        <CardTitle className="text-xl">{dossier.title}</CardTitle>
      </CardHeader>
      <CardContent className="grid gap-5">
        <Section title="People">
          <div className="grid gap-3">
            {dossier.attendees.map((attendee) => (
              <div key={attendee.email ?? attendee.name}>
                <p className="font-medium">{attendee.name}</p>
                <p className="text-sm text-muted-foreground">{[attendee.company, attendee.email].filter(Boolean).join(" · ")}</p>
                {attendee.notes ? <p className="mt-1 text-sm">{attendee.notes}</p> : null}
              </div>
            ))}
          </div>
        </Section>
        <Section title="Context"><p className="whitespace-pre-wrap text-sm leading-6">{dossier.context}</p></Section>
        <Section title="Have ready">
          <ul className="list-disc space-y-1 pl-5 text-sm">{dossier.prepare.map((item) => <li key={item}>{item}</li>)}</ul>
        </Section>
        {dossier.threads.length === 0 ? null : (
          <Section title="Recent threads">
            <div className="grid gap-3">{dossier.threads.map((thread) => <div key={thread.subject}><p className="font-medium">{thread.subject}</p><p className="text-sm text-muted-foreground">{thread.takeaway}</p></div>)}</div>
          </Section>
        )}
      </CardContent>
    </Card>
  )
}

function Section({ children, title }: { children: React.ReactNode; title: string }) {
  return <section className="grid gap-3"><Separator /><h2 className="text-sm font-semibold">{title}</h2>{children}</section>
}

function useDossiers() {
  return useMiloState(contract.state.dossiers)
}

function readLocation() {
  const params = new URLSearchParams(window.location.hash.replace(/^#/, ""))
  return { day: params.get("d") ?? localDate(new Date()), meetingId: params.get("m") }
}

function localDate(date: Date) {
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, "0")
  const day = String(date.getDate()).padStart(2, "0")
  return `${year}-${month}-${day}`
}

function attendeeName(attendee: Dossier["attendees"][number]) {
  return attendee.company ? `${attendee.name} (${attendee.company})` : attendee.name
}

function formatTime(value: string) {
  return new Intl.DateTimeFormat(undefined, { hour: "2-digit", minute: "2-digit" }).format(new Date(value))
}

function formatDateTime(value: string) {
  return new Intl.DateTimeFormat(undefined, { dateStyle: "medium", timeStyle: "short" }).format(new Date(value))
}

function isNext(startsAt: string, meetings: Array<[string, Dossier]>) {
  const now = Date.now()
  return meetings.find(([, dossier]) => new Date(dossier.startsAt).getTime() >= now)?.[1].startsAt === startsAt
}
