import { z } from "zod"
import { defineArtifactContract } from "@/milo/contract"

const attendee = z.strictObject({
  name: z.string(),
  email: z.string().optional(),
  company: z.string().optional(),
  notes: z.string().optional(),
})

const dossier = z.strictObject({
  title: z.string(),
  startsAt: z.iso.datetime(),
  preparedAt: z.iso.datetime(),
  attendees: z.array(attendee),
  context: z.string(),
  prepare: z.array(z.string()),
  threads: z.array(z.strictObject({ subject: z.string(), takeaway: z.string() })),
})

export const contract = defineArtifactContract({
  version: 1,
  state: {
    dossiers: {
      key: "prep/dossiers",
      scope: "shared",
      schema: z.strictObject({
        schemaVersion: z.literal(1),
        days: z.record(z.string(), z.record(z.string(), dossier)),
      }),
      schemaName: "MeetingPrepDossiers",
      schemaVersion: 1,
    },
  },
})
